#include <iostream>
#include <vector>
#include <algorithm>
#include <set>
#define ll long long int
using namespace std;

void solve(){
    int CH,BX;
    cin>>CH>>BX;
    vector<pair<int,int>> wv;
    vector<int>chl(CH+1),bxl(BX+1);
    for(int i=1;i<=CH;i++) {
        int chocw;
        cin>>chocw;
        wv.emplace_back(make_pair(chocw,-i));
    }
    for(int i=1;i<=CH;i++) {
        cin>>chl[i];
    }
    for(int i=1;i<=BX;i++) {
        int boxw;
        cin>>boxw;
        wv.emplace_back(make_pair(boxw,i));
    }
    for(int i=1;i<=BX;i++) {
        cin>>bxl[i];
    }
    sort(wv.begin(),wv.end(),greater<>());
    multiset<int>ms;
    for (auto[w,idx]:wv) {
        if (idx>0) {
            ms.insert(bxl[idx]);
        }else {
            auto it = ms.lower_bound(chl[-idx]);
            if (it!=ms.end()) {
                ms.erase(it);
            }else {
                cout<<"NO"; return;
            }
        }
    }
    cout<<"YES";
}


int main () {
    ios_base::sync_with_stdio (false);
    cin.tie(nullptr);
    int T=1;
    while (T--) {
        solve();
    }
}
