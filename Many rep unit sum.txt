#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;

void solve(){
  int N;
  cin >> N;
  vector<int> a(N), ans;
  for (int i = 0; i < N; i++) cin >> a[i];
  sort(a.begin(), a.end());
  ll cfwd = 0;
  for (int ctr = 1; ctr <= a.back(); ctr++){
    auto it = lower_bound(a.begin(), a.end(), ctr);
    ll cnt = a.end() - it;
    cfwd += cnt;
    ans.push_back(cfwd % 10);
    cfwd = cfwd / 10;
  }
  while (cfwd != 0){
    ans.push_back(cfwd % 10);
    cfwd = cfwd / 10;
  }
  for (int i = (int)ans.size() - 1; i >= 0; i--) cout << ans[i];
  cout << "\n";
}

int main(){
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  solve();
  return 0;
}
