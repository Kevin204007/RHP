#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int hcf(int a, int b) {
    return b==0?a:hcf(b,a%b);
}

void solve(){
    int N; cin>>N;
    vector<int> a(N);
    for (int i=0;i<N;i++) {
        cin>>a[i];
    }
    cout<<hcf(a[0],a[N-1])<<endl;
}


int main () {
    ios_base::sync_with_stdio (false);
    cin.tie(nullptr);
    int T;cin>>T;
    while (T--) {
        solve();
    }
}
