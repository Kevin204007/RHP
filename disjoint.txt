#include <iostream>
#include <vector>
#include <algorithm>
#include <set>
#define ll long long int
using namespace std;

int find(vector<int>&ldr,int node) {
    if (ldr[node]!=node) {
        ldr[node]=find(ldr,ldr[node]);
    }
    return ldr[node];
}

void join(vector<int>&ldr,int lt,int rt) {
    int ldrlt=find(ldr,lt);
    int ldrrt=find(ldr,rt);
    ldr[ldrrt]=ldrlt;
}

void solve () {
    int N,M;
    cin >> N>>M;
    vector<int> ldr(N+1);
    for (int i = 1; i <= N; i++) {
        ldr[i]=i;
    }
    for (int j = 1; j <= M; j++) {
        int lt,rt;
        cin >> lt >> rt;
        join(ldr,lt,rt);
    };
    set<int> st;
    for (int i = 1; i <= N; i++) {
        st.insert(find(ldr,i)); //lazy leader update so ldr[i]
    }
    cout<<st.size();
}

int main () {
    ios_base::sync_with_stdio (false);
    cin.tie(nullptr);
    int T=1;
    while (T--) {
        solve();
    }
}
