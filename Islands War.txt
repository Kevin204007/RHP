#include<bits/stdc++.h>
using namespace std;

void solve(){
  int N,Q; cin>>N>>Q;
  vector<pair<int,int>> arr(Q);
  for(int i=0;i<Q;i++){
    int lt,rt;cin>>lt>>rt;
    arr[i]={rt,lt};
  }
  sort(arr.begin(),arr.end());
  int count=0,rtmax=-1;
  for(auto[rt,lt]:arr){
    if(lt>=rtmax){
      count++;
      rtmax=rt;
    }
  }
  cout<<count<<endl;
}

int main (){
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int T=1;
  //cin>>T;
  while(T--) solve();
}
