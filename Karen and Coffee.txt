import java.util.*;

public class Main {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();

        int m = 200000;

        int[] d = new int[m + 2];

        for (int i = 0; i < a; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();

            d[x]++;
            d[y + 1]--;
        }

        int[] f = new int[m + 1];
        f[1] = d[1];

        for (int i = 2; i <= m; i++) {
            f[i] = f[i - 1] + d[i];
        }

        int[] g = new int[m + 1];

        for (int i = 1; i <= m; i++) {
            if (f[i] >= b) {
                g[i] = 1;
            }
        }

        int[] p = new int[m + 1];
        p[1] = g[1];

        for (int i = 2; i <= m; i++) {
            p[i] = p[i - 1] + g[i];
        }

        while (c-- > 0) {
            int l = sc.nextInt();
            int r = sc.nextInt();

            System.out.println(p[r] - p[l - 1]);
        }
    }
}
