class MinStack {
public:
    vector<pair<int,int>> st;
    MinStack() {
        
    }
    
    void push(int value) {
        int target= value;
        if(!st.empty()){
            target=min(value,getMin());
        }
        st.push_back(make_pair(value,target));

    }
    
    void pop() {
        st.pop_back();
    }
    
    int top() {
        return st.back().first;
    }
    
    int getMin() {
        return st.back().second;
    }
};
