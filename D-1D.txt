#include <iostream>
#include <vector>
#include <algorithm>
#define ll long long int
using namespace std;

void solve() {
    int n;
    cin >> n;
    vector<int> loc(n);
    vector<ll> vpsum(n+1,0LL);
    for (int i=0; i<n; i++) {
        cin >> loc[i];
    }
    for (int i=1; i<=n; i++) {
        int ppl;
        cin >> ppl;
        vpsum[i] = vpsum[i-1] + ppl;
    }
    int q;
    cin >> q;
    while (q--) {
        int ql,qr;
        cin >>ql >> qr;
        int lt=lower_bound(loc.begin(),loc.end(),ql)-loc.begin();
        int rt=upper_bound(loc.begin(),loc.end(),qr)-loc.begin();
        cout<<vpsum[rt]-vpsum[lt]<<endl;
    }
}
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int q=1;
    while (q--) {
        solve();
    }

}
