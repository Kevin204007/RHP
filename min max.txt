#include <iostream>
#include <vector>
using namespace std;
#define ll long long

void solve() {
    int N;
    cin >> N;
    vector<int> a(N + 1);
    for (int i = 1; i <= N; i++) cin >> a[i];

    ll same = 0, exchg = 0;
    for (int i = 1; i <= N; i++) {
        if (a[i] == i) {
            same++;
        } else if (a[a[i]] == i) {
            exchg++;
        }
    }

    ll ans = exchg / 2 + same * (same - 1) / 2;
    cout << ans << endl;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    solve();
}
