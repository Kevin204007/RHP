#include <iostream>
using namespace std;

int main() {
    int r, c;
    cin >> r >> c;

    int grid[r][c];
    int dp[r][c];

    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++) {
            cin >> grid[i][j];
        }
    }

    for (int j = 0; j < c; j++) {
        dp[0][j] = grid[0][j];
    }

    for (int i = 1; i < r; i++) {
        for (int j = 0; j < c; j++) {

            int maxi = 0;

            for (int k = 0; k < c; k++) {
                if (k != j) {
                    maxi = max(maxi, dp[i - 1][k]);
                }
            }

            dp[i][j] = grid[i][j] + maxi;
        }
    }

    int ans = 0;

    for (int j = 0; j < c; j++) {
        ans = max(ans, dp[r - 1][j]);
    }

    cout << ans;

    return 0;
}
